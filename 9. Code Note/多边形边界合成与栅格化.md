#### 2. 栅格化gpkg数据

``` python
from rasterio.transform import from_origin  
import rasterio.features
import geopandas as gpd  
import numpy as np

gdf_o_union_mare = gpd.read_file(r"C:\Users\caiji\Desktop\merged_mare.gpkg")  
  
minx, miny, maxx, maxy = -180, -90, 180, 90  
width = maxx - minx  # 经度范围  
height = maxy - miny  # 纬度范围  
# 定义仿射变换（地理坐标 → 像素坐标）  
transform = from_origin(  
    minx, maxy,  # 左上角坐标  
    width / 2000,  # 像素宽度（经度分辨率）  
    height / 2000   # 像素高度（纬度分辨率）  
)  
  
# 栅格化（多边形内填充 1）  
rasterized = rasterio.features.rasterize(  
    shapes=gdf_o_union_mare.geometry,  # 要栅格化的几何  
    out_shape=(2000, 2000),  
    transform=transform,  
    fill=0,      # 背景值  
    default_value=1  # 多边形内值  
)  
matrix = rasterized.astype(np.uint8)  # 转换为 0-1 矩阵  
np.save("resource/mare_matrix.npy", matrix)
```
